    30→function formatBytes(n) {
    31→  if (!n) return '0 KB';
    32→  const kb = n / 1024;
    33→  return kb < 1024 ? `${kb.toFixed(0)} KB` : `${(kb / 1024).toFixed(1)} MB`;
    34→}

    60→  function myPagesForAttachment(attachmentId) {
    61→    const rows = (assignmentsByAttachment[attachmentId] || []).filter((r) => r.assignedTo?.id === user?.id);
    62→    if (!rows.length) return undefined; // not assigned to me at all
    63→    if (rows.some((r) => !r.pageNumbers || !r.pageNumbers.length)) return null; // whole doc
    64→    const set = new Set();
    65→    rows.forEach((r) => (r.pageNumbers || []).forEach((p) => set.add(p)));
    66→    return [...set].sort((a, b) => a - b);
    67→  }
    68→
    69→  // total pages (client-detected via pdf.js on thumbnail load) vs. how many
    70→  // are done / actively being worked on / just handed out / still free —
    71→  // this is the "kitna baki, kitna hua" breakdown for PE/Dept Head/Plant
    72→  // Head/Admin oversight, not just a plain assigned-vs-not count.
    73→  function pageCoverage(attachmentId) {
    74→    const total = numPagesByAttachment[attachmentId];
    75→    const rows = assignmentsByAttachment[attachmentId] || [];
    76→    const completedSet = new Set();
    77→    const inProgressSet = new Set();
    78→    const assignedSet = new Set(); // handed out but not started (ASSIGNED/REOPENED)
    79→    for (const r of rows) {
    80→      const pages = r.pageNumbers?.length ? r.pageNumbers : (total ? Array.from({ length: total }, (_, i) => i + 1) : []);
    81→      const bucket = r.status === 'COMPLETED' ? completedSet : r.status === 'IN_PROGRESS' ? inProgressSet : assignedSet;
    82→      pages.forEach((p) => bucket.add(p));
    83→    }
    84→    // a page can appear in more than one bucket if it was re-delegated at
    85→    // different stages — completed wins, then in-progress, then assigned
    86→    for (const p of completedSet) { inProgressSet.delete(p); assignedSet.delete(p); }
    87→    for (const p of inProgressSet) assignedSet.delete(p);
    88→    const takenCount = completedSet.size + inProgressSet.size + assignedSet.size;
    89→    const remaining = total
    90→      ? Array.from({ length: total }, (_, i) => i + 1).filter((p) => !completedSet.has(p) && !inProgressSet.has(p) && !assignedSet.has(p))
    91→      : [];
    92→    return {
    93→      total, remaining,
    94→      completedCount: completedSet.size, inProgressCount: inProgressSet.size, assignedCount: assignedSet.size,
    95→      takenCount,
    96→    };
    97→  }
    98→
    99→  // { [pageNumber]: "Department (User)" } for every page already handed out —
   100→  // passed into the PDF viewer's select mode so those pages show locked and
   101→  // can't be assigned a second time by accident.
   102→  function takenPagesFor(attachmentId) {
   103→    const total = numPagesByAttachment[attachmentId];
   104→    const rows = assignmentsByAttachment[attachmentId] || [];
   105→    const map = {};
   106→    for (const r of rows) {
   107→      const label = `${r.department?.name || '—'} (${r.assignedTo?.name || '—'})`;
   108→      if (r.pageNumbers?.length) {
   109→        r.pageNumbers.forEach((p) => { map[p] = label; });
   110→      } else if (total) {
   111→        for (let p = 1; p <= total; p++) map[p] = label;
   112→      }
   113→    }
   114→    return map;
   115→  }

   138→  async function loadAssignments(list) {

   140→    if (!pdfIds.length) return;

   145→      setAssignmentsByAttachment(Object.fromEntries(results));
   146→    } catch {
   147→      // non-critical — cards just show without the assignment summary
   148→    }
   149→  }

   151→  async function fetchBlobUrl(a) {
   152→    const r = await api.get(`/attachments/file/${a.id}`, { responseType: 'blob' });
   153→    return URL.createObjectURL(new Blob([r.data], { type: a.mimeType }));
   154→  }
   155→
   156→  async function downloadFile(a) {
   157→    const url = await fetchBlobUrl(a);
   158→    const link = document.createElement('a');
   159→    link.href = url; link.download = a.filename;
   160→    document.body.appendChild(link); link.click(); link.remove();
   161→    URL.revokeObjectURL(url);
   162→  }
   163→
   164→  async function openImage(a) {

   166→    setLightbox({ url, alt: a.filename });

   169→  async function deleteAttachment(a) {
   170→    if (!confirm(`Permanently delete "${a.filename}"? This cannot be undone.`)) return;
   171→    try {
   172→      await api.delete(`/attachments/file/${a.id}`);
   173→      toast.success('File deleted');

   175→    } catch (err) {
   176→      console.error(err);
   177→      toast.error(err.response?.data?.message || 'Failed to delete file');
   178→    }
   179→  }

   181→  async function uploadGeneralFiles(e) {
   182→    const files = Array.from(e.target.files || []);
   183→    if (!files.length) return;

   185→    files.forEach((f) => fd.append('files', f));

   188→      await api.post(`/attachments/JOBCARD/${id}`, fd);

   312→        <div className="grid grid-cols-3 gap-4">
   313→          {[

   315→            { label: 'Produced', value: Number(jc.qtyProduced), color: 'text-emerald-600' },

   320→              <div className="text-xs text-slate-500 font-medium uppercase tracking-wider mt-1">{s.label}</div>
   321→            </div>
   322→          ))}
   323→        </div>
   324→
   325→        <div>
   326→          <div className="flex items-center gap-2 mb-3">
   327→            <Package className="w-4 h-4 text-brand-600" />
   328→            <div className="font-semibold text-sm text-slate-800">Materials</div>
   329→          </div>
   330→          <div className="overflow-x-auto">
   331→            <table className="min-w-full">
   332→              <thead><tr><th className="table-th rounded-tl-lg">Item</th><th className="table-th">Qty</th><th className="table-th">UoM</th><th className="table-th rounded-tr-lg">Notes</th></tr></thead>
   333→              <tbody>

   555→            </div>
   556→          </div>
   557→        )}
   558→      </div>
   559→    </div>
   560→  );
   561→}
   562→
   563→function ImageThumb({ attachment, onOpen }) {
   564→  const [url, setUrl] = useState('');
   565→  useEffect(() => {
   566→    let cancelled = false;
   567→    let objUrl;
   568→    api.get(`/attachments/file/${attachment.id}`, { responseType: 'blob' }).then((r) => {
   569→      if (cancelled) return;
   570→      objUrl = URL.createObjectURL(r.data);
   571→      setUrl(objUrl);
   572→    });
   573→    return () => { cancelled = true; if (objUrl) URL.revokeObjectURL(objUrl); };
   574→  }, [attachment.id]);
   575→
   576→  return (
   577→    <button
   578→      type="button"
   579→      onClick={onOpen}
   580→      className="aspect-square rounded-lg overflow-hidden border border-slate-200 bg-slate-50 hover:opacity-80 transition-opacity"
   581→      title={attachment.filename}
   582→    >
   583→      {url ? <img src={url} alt={attachment.filename} className="w-full h-full object-cover" /> : null}
   584→    </button>
   585→  );
   586→}
