    30→function formatBytes(n) {
    31→  if (!n) return '0 KB';
    32→  const kb = n / 1024;
    33→  return kb < 1024 ? `${kb.toFixed(0)} KB` : `${(kb / 1024).toFixed(1)} MB`;
    34→}

    61→  function myPagesForAttachment(attachmentId) {
    62→    const rows = (assignmentsByAttachment[attachmentId] || []).filter((r) => r.assignedTo?.id === user?.id);
    63→    if (!rows.length) return undefined; // not assigned to me at all
    64→    if (rows.some((r) => !r.pageNumbers || !r.pageNumbers.length)) return null; // whole doc
    65→    const set = new Set();
    66→    rows.forEach((r) => (r.pageNumbers || []).forEach((p) => set.add(p)));
    67→    return [...set].sort((a, b) => a - b);
    68→  }
    69→
    70→  // total pages (client-detected via pdf.js on thumbnail load) vs. how many
    71→  // are done / actively being worked on / just handed out / still free —
    72→  // this is the "kitna baki, kitna hua" breakdown for PE/Dept Head/Plant
    73→  // Head/Admin oversight, not just a plain assigned-vs-not count.
    74→  function pageCoverage(attachmentId) {
    75→    const total = numPagesByAttachment[attachmentId];
    76→    const rows = assignmentsByAttachment[attachmentId] || [];
    77→    const completedSet = new Set();
    78→    const inProgressSet = new Set();
    79→    const assignedSet = new Set(); // handed out but not started (ASSIGNED/REOPENED)
    80→    for (const r of rows) {
    81→      const pages = r.pageNumbers?.length ? r.pageNumbers : (total ? Array.from({ length: total }, (_, i) => i + 1) : []);
    82→      const bucket = r.status === 'COMPLETED' ? completedSet : r.status === 'IN_PROGRESS' ? inProgressSet : assignedSet;
    83→      pages.forEach((p) => bucket.add(p));
    84→    }
    85→    // a page can appear in more than one bucket if it was re-delegated at
    86→    // different stages — completed wins, then in-progress, then assigned
    87→    for (const p of completedSet) { inProgressSet.delete(p); assignedSet.delete(p); }
    88→    for (const p of inProgressSet) assignedSet.delete(p);
    89→    const takenCount = completedSet.size + inProgressSet.size + assignedSet.size;
    90→    const remaining = total
    91→      ? Array.from({ length: total }, (_, i) => i + 1).filter((p) => !completedSet.has(p) && !inProgressSet.has(p) && !assignedSet.has(p))
    92→      : [];
    93→    return {
    94→      total, remaining,
    95→      completedCount: completedSet.size, inProgressCount: inProgressSet.size, assignedCount: assignedSet.size,
    96→      takenCount,
    97→    };
    98→  }
    99→
   100→  // { [pageNumber]: "Department (User)" } for every page already handed out —
   101→  // passed into the PDF viewer's select mode so those pages show locked and
   102→  // can't be assigned a second time by accident.
   103→  function takenPagesFor(attachmentId) {
   104→    const total = numPagesByAttachment[attachmentId];
   105→    const rows = assignmentsByAttachment[attachmentId] || [];
   106→    const map = {};
   107→    for (const r of rows) {
   108→      const label = `${r.department?.name || '—'} (${r.assignedTo?.name || '—'})`;
   109→      if (r.pageNumbers?.length) {
   110→        r.pageNumbers.forEach((p) => { map[p] = label; });
   111→      } else if (total) {
   112→        for (let p = 1; p <= total; p++) map[p] = label;
   113→      }
   114→    }
   115→    return map;
   116→  }

   118→  async function load() {

   120→      const r = await api.get(`/jobcards/${id}`);

   137→  // which pages of each PDF have already been assigned, and to which

   139→  async function loadAssignments(list) {

   141→    if (!pdfIds.length) return;

   144→        pdfIds.map((aid) => api.get('/assignments', { params: { attachmentId: aid } }).then((r) => [aid, r.data]))
   145→      );
   146→      setAssignmentsByAttachment(Object.fromEntries(results));
   147→    } catch {
   148→      // non-critical — cards just show without the assignment summary
   149→    }
   150→  }

   152→  async function fetchBlobUrl(a) {
   153→    const r = await api.get(`/attachments/file/${a.id}`, { responseType: 'blob' });
   154→    return URL.createObjectURL(new Blob([r.data], { type: a.mimeType }));
   155→  }
   156→
   157→  async function downloadFile(a) {
   158→    const url = await fetchBlobUrl(a);
   159→    const link = document.createElement('a');
   160→    link.href = url; link.download = a.filename;
   161→    document.body.appendChild(link); link.click(); link.remove();
   162→    URL.revokeObjectURL(url);
   163→  }
   164→
   165→  async function openImage(a) {

   168→  }

   170→  async function deleteAttachment(a) {

   172→    try {
   173→      await api.delete(`/attachments/file/${a.id}`);

   176→    } catch (err) {
   177→      console.error(err);
   178→      toast.error(err.response?.data?.message || 'Failed to delete file');
   179→    }
   180→  }

   182→  async function uploadGeneralFiles(e) {
   183→    const files = Array.from(e.target.files || []);
   184→    if (!files.length) return;
   185→    const fd = new FormData();
   186→    files.forEach((f) => fd.append('files', f));

   327→        </div>
   328→
   329→        <div>

   559→            </div>
   560→          </div>
   561→        )}
   562→      </div>
   563→    </div>
   564→  );
   565→}

   567→function ImageThumb({ attachment, onOpen }) {
   568→  const [url, setUrl] = useState('');
   569→  useEffect(() => {
   570→    let cancelled = false;
   571→    let objUrl;
   572→    api.get(`/attachments/file/${attachment.id}`, { responseType: 'blob' }).then((r) => {
   573→      if (cancelled) return;
   574→      objUrl = URL.createObjectURL(r.data);
   575→      setUrl(objUrl);
   576→    });
   577→    return () => { cancelled = true; if (objUrl) URL.revokeObjectURL(objUrl); };
   578→  }, [attachment.id]);
   579→
   580→  return (
   581→    <button
   582→      type="button"
   583→      onClick={onOpen}
   584→      className="aspect-square rounded-lg overflow-hidden border border-slate-200 bg-slate-50 hover:opacity-80 transition-opacity"
   585→      title={attachment.filename}
   586→    >
   587→      {url ? <img src={url} alt={attachment.filename} className="w-full h-full object-cover" /> : null}
   588→    </button>
   589→  );
   590→}
