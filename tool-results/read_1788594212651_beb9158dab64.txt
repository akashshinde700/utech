    63→model Role {
    64→  id                 Int              @id @default(autoincrement())
    65→  name               String           @unique
    66→  // short unique machine code (e.g. "DEPARTMENT_HEAD") — distinct from the
    67→  // free-text display `name`, required for roles created via the quick-add
    68→  // popup but nullable so pre-existing hand-seeded roles aren't broken
    69→  code               String?          @unique
    70→  description        String?
    71→  isSystem           Boolean          @default(false)
    72→  isActive           Boolean          @default(true)
    73→  // numeric rank (0 = highest, e.g. Super Admin) — drives the "Reporting To"
    74→  // dropdown generically: candidates for a role are users whose role has a
    75→  // lower hierarchyLevel, with no hardcoded role-name-to-role-name map.
    76→  hierarchyLevel     Int?
    77→  // organizational parent in the Role Master tree (display/reference only —
    78→  // does not itself drive any permission or scoping logic)
    79→  parentRoleId       Int?
    80→  parentRole         Role?            @relation("RoleParent", fields: [parentRoleId], references: [id])
    81→  childRoles         Role[]           @relation("RoleParent")
    82→  // when true, the Users form requires a Department for this role
    83→  requiresDepartment Boolean          @default(false)
    84→  // when true, users with this role only see/manage same-department users
    85→  // in the Users module (Department Head / Supervisor / Team Leader)
    86→  scopeToDepartment  Boolean          @default(false)
    87→  permissions        RolePermission[]
    88→  users              User[]
    89→  createdAt          DateTime         @default(now())
    90→  updatedAt          DateTime         @updatedAt
    91→}

   904→// The Project Engineer assigns a project document to a department; when that
   905→// department (in practice Vendor Development) cannot do the work in-house it
   906→// hands the scope to an external vendor. This model is that hand-off: it
   907→// answers "when did it go out, who did we give it to, and did it come back".
   908→//
   909→// It is deliberately a *tracker*, not a stock document — a vendor very often
   910→// manufactures from its own raw material, so there is nothing to move. Stock is
   911→// always moved by exactly one linked document instead:
   912→//   - material issued to the vendor  -> jobworkChallan (stock OUT)
   913→//   - finished parts received back    -> grns          (stock IN)
   914→// and once poId is set, GRN is the only way quantities can be received, so the
   915→// tracker and the stock documents can never drift apart.
   916→model VendorWorkOrder {
   917→  id     Int    @id @default(autoincrement())

   919→
   920→  // --- where the work came from ---

   924→  jobcard      Jobcard?    @relation(fields: [jobcardId], references: [id])
   925→  departmentId Int?
   926→  department   Department? @relation(fields: [departmentId], references: [id])
   927→
   928→  // --- who it went out to ---
   929→  partyId Int
   930→  party   Party @relation("PartyVendorWorkOrders", fields: [partyId], references: [id])
   931→
   932→  status VendorWorkOrderStatus @default(DRAFT)
